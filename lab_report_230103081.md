## Verification and reproducibility

- Each full timing configuration uses two warm-ups and seven samples; median and IQR are stored in JSON.
- Process workers are top-level functions so macOS 'spawn' multiprocessing can import them.
- Parallel calculations use OS processes; the false-sharing, memory-copy and GEMM experiments use SharedMemory.
- Scaling also reads SharedMemory in the pool while leaving the shared buffer read-only.
- All numerical results must come from an actual local run; no benchmark timings are hard-coded.
- The run is hardware-specific; excessive CPU load, thermal throttling and startup overhead affect measurements.

@dataclass
class BenchStats:
    min_ms: float
    median_ms: float
    iqr_ms: float
    samples_ms: list[float]


def stats_from_samples(samples: list[float]) -> BenchStats:
    if not samples:
        raise ValueError("No timing samples")
    ordered = sorted(samples)
    n = len(ordered)
    return BenchStats(ordered[0], ordered[n // 2],
                      ordered[(3 * n) // 4] - ordered[n // 4], ordered)


def run_benchmark(fn: Callable[[], None], warmup: int = 2, runs: int = 7) -> BenchStats:
    for _ in range(warmup):
        fn()
    samples = []
    for _ in range(runs):
        t0 = time.perf_counter_ns()
        fn()
        samples.append((time.perf_counter_ns() - t0) / 1e6)
    return stats_from_samples(samples)


def fnv1a_hash(s: str) -> int:
    h = 14695981039346656037
    for b in s.encode("utf-8"):

    return h


def emit_verify_tag(phase: str, seed: int, m1: float, m2: float) -> str:

def spawn_context():
    # Explicit spawn works on macOS and Linux, including on Apple Silicon.
    return mp.get_context("spawn")


def partitions(total: int, workers: int) -> list[tuple[int, int]]:
    q, r = divmod(total, workers)
    out, start = [], 0
    for rank in range(workers):
        end = start + q + (1 if rank < r else 0)
        out.append((start, end))
        start = end
    return out


def geometric_counts(logical: int) -> list[int]:
    result, p = [], 1
    while p <= logical:
        result.append(p)
        p *= 2
    if logical not in result:
        result.append(logical)
    return sorted(set(result))


def _sysctl(key: str) -> str | None:
    try:
        p = subprocess.run(["sysctl", "-n", key], capture_output=True,
                           text=True, check=False, timeout=8)
        return p.stdout.strip() if p.returncode == 0 and p.stdout.strip() else None
    except (OSError, subprocess.TimeoutExpired):
        return None


def _as_int(v: str | None) -> int | None:
    if v is None:
        return None
    try:
        n = int(v)
        return n if n >= 0 else None
    except ValueError:
        return None


def _linux_cpu_cache() -> dict:
    result = {"line_bytes": None, "l1d_bytes": None,
              "l2_bytes": None, "l3_bytes": None}
    for base in sorted(Path("/sys/devices/system/cpu/cpu0/cache").glob("index*")):
        try:
            typ = (base / "type").read_text().strip().lower()
            level = int((base / "level").read_text().strip())
            size = (base / "size").read_text().strip().upper()
            m = re.match(r"(\d+)([KMG]?)", size)
            if not m:
                continue
            sz = int(m.group(1)) * {"": 1, "K": 1024, "M": 1024**2, "G": 1024**3}[m.group(2)]
            if typ in ("data", "unified"):
                if level == 1 and typ == "data":
                    result["l1d_bytes"] = sz
                elif level == 2:
                    result["l2_bytes"] = sz
                elif level == 3:
                    result["l3_bytes"] = sz
            result["line_bytes"] = int((base / "coherency_line_size").read_text().strip())
        except (OSError, ValueError):
            continue
    return result


def hardware_info() -> dict:
    sysname = platform.system()
    logical = mp.cpu_count()
    info = {"system": sysname, "cpu_model": platform.processor() or "Unknown",
            "physical_cores": None, "logical_threads": logical,
            "line_bytes": None, "l1d_bytes": None, "l2_bytes": None,
            "l3_bytes": None, "notes": []}
    if sysname == "Darwin":
        info["cpu_model"] = _sysctl("machdep.cpu.brand_string") or _sysctl("hw.model") or info["cpu_model"]
        if info["cpu_model"].startswith("Mac"):
            try:
                sp = subprocess.run(["system_profiler", "SPHardwareDataType"], capture_output=True,
                                    text=True, timeout=15, check=False)
                for line in sp.stdout.splitlines():
                    if line.strip().startswith("Chip:"):
                        info["cpu_model"] = line.partition(":")[2].strip()
                        break
            except (OSError, subprocess.TimeoutExpired):
                pass
        info["physical_cores"] = _as_int(_sysctl("hw.physicalcpu"))
        info["logical_threads"] = _as_int(_sysctl("hw.logicalcpu")) or logical
        for name, key in (("line_bytes", "hw.cachelinesize"),
                          ("l1d_bytes", "hw.l1dcachesize"),
                          ("l2_bytes", "hw.l2cachesize"),
                          ("l3_bytes", "hw.l3cachesize")):
            info[name] = _as_int(_sysctl(key))
        info["notes"].append("On heterogeneous Apple CPUs, sysctl cache sizes may describe one core/cluster; L3 can be absent.")
    elif sysname == "Linux":
        try:
            lines = Path("/proc/cpuinfo").read_text().splitlines()
            for line in lines:
                if line.startswith("model name"):
                    info["cpu_model"] = line.partition(":")[2].strip()
                    break
        except OSError:
            pass
        info.update(_linux_cpu_cache())
        try:
            pairs = set()
            for path in Path("/sys/devices/system/cpu").glob("cpu[0-9]*/topology/physical_package_id"):
                cpu_dir = path.parent
                pairs.add((path.read_text().strip(), (cpu_dir / "core_id").read_text().strip()))
            info["physical_cores"] = len(pairs) if pairs else None
        except OSError:
            pass
        info["notes"].append("Linux cache sizes are sourced from cpu0 sysfs; cgroup CPU quotas may restrict usable cores.")
    elif sysname == "Windows":
        try:
            cmd = "Get-CimInstance Win32_Processor | ConvertTo-Json -Compress"
            data = subprocess.run(["powershell", "-NoProfile", "-Command", cmd],
                                  capture_output=True, text=True, timeout=15, check=False)
            vals = json.loads(data.stdout)
            if isinstance(vals, dict):
                vals = [vals]
            info["physical_cores"] = sum(int(v.get("NumberOfCores") or 0) for v in vals) or None
            info["cpu_model"] = ", ".join(str(v.get("Name")) for v in vals)
            info["l2_bytes"] = sum(int(v.get("L2CacheSize") or 0) for v in vals) * 1024 or None
            info["l3_bytes"] = sum(int(v.get("L3CacheSize") or 0) for v in vals) * 1024 or None
        except (OSError, ValueError, subprocess.TimeoutExpired):
            pass
    if not info["physical_cores"]:
        # os.cpu_count is only a fallback, and must be clearly labelled.
        info["physical_cores"] = logical
        info["notes"].append("Physical core detection failed; logical CPU count used as a fallback. Verify manually.")
    return info

"""Phase 1: true multi-process false-sharing strides and Karp-Flatt diagnostics."""
import argparse
import ctypes
from multiprocessing import shared_memory
from harness import (emit_verify_tag, geometric_counts, hardware_info,
                     package_stats, process_counts, run_benchmark, save_json,
                     settings, spawn_context, partitions)

UINT64_MASK = 0xFFFFFFFFFFFFFFFF
SHARED_BYTES = 4096


def fs_worker(shm_name: str, slot_idx: int, iters: int) -> None:
    shm = shared_memory.SharedMemory(name=shm_name)
    arr = (ctypes.c_uint64 * 512).from_buffer(shm.buf)
    try:
        acc = arr[slot_idx]
        for i in range(iters):
            acc = (acc * 6364136223846793005 + (i | 1)) & UINT64_MASK
            arr[slot_idx] = acc
    finally:
        del arr
        shm.close()


def run_case(p: int, stride: int, operations: int, warmup: int, runs: int) -> dict:
    if (p - 1) * stride >= 512:
        raise ValueError(f"4096-byte block cannot fit p={p}, stride={stride}; 512 uint64 slots available")
    ctx = spawn_context()
    shm = shared_memory.SharedMemory(create=True, size=SHARED_BYTES)
    positions = partitions(operations, p)
    def trial() -> None:
        # macOS (including Apple Silicon) can round 4096 bytes up to a
        # 16384-byte shared-memory page. Only the requested logical 4096-byte
        # region is used; any larger OS allocation is entirely acceptable.
        if len(shm.buf) < SHARED_BYTES:
            raise RuntimeError(f"SharedMemory is only {len(shm.buf)} bytes; need at least {SHARED_BYTES}")
        # Clear exactly the logical 4096-byte region, avoiding memoryview
        # slice-assignment differences across Python/macOS builds.
        ctypes.memset(ctypes.addressof(ctypes.c_char.from_buffer(shm.buf)),
                      0, SHARED_BYTES)
        children = [ctx.Process(target=fs_worker, args=(shm.name, rank * stride, end - start))
                    for rank, (start, end) in enumerate(positions)]
        for child in children:
            child.start()
        for child in children:
            child.join()
        if any(c.exitcode != 0 for c in children):
            raise RuntimeError("A false-sharing worker exited abnormally")
    try:
        stats = run_benchmark(trial, warmup, runs)
    finally:
        shm.close()
        shm.unlink()
    return {"p": p, "stride": stride, "stride_bytes": stride * 8,
            "operations": operations, "throughput_mops": operations / (stats.median_ms * 1000.0),
            **package_stats(stats)}


def run(student_id: int, mode: str = "full") -> dict:
    hw = hardware_info()
    cfg = settings(mode)
    physical, logical = process_counts(hw["logical_threads"], hw["physical_cores"], mode)
    if physical <= 0:
        raise ValueError("Invalid physical CPU count")
    cases = {}
    def get(p: int, stride: int) -> dict:
        key = (p, stride)
        if key not in cases:
            case = run_case(p, stride, cfg["phase1_ops"], cfg["warmup"], cfg["runs"])
            cases[key] = case
            print(f"p={p:>2}, stride={stride:>2}: {case['median_ms']:.3f} ms, {case['throughput_mops']:.3f} Mops/s", flush=True)
        return cases[key]
    stride_sweep = [get(physical, stride) for stride in cfg["strides"]]
    scaling = []
    for stride in (1, 16):
        t1 = get(1, stride)["median_ms"]
        for p in geometric_counts(logical):
            item = dict(get(p, stride))
            sp = t1 / item["median_ms"]
            item["speedup"] = sp
            item["efficiency"] = sp / p

    if improvements:
        best_change = max(improvements, key=lambda x: x["relative_speed_gain"])
    median1 = get(physical, 1)["median_ms"]
    median16 = get(physical, 16)["median_ms"]
    tag = emit_verify_tag("PHASE1", student_id, median1, median16)
    print(tag)

"""Phase 2: real process-pool strong and weak scaling with a serial stage."""
import argparse
import ctypes
import math
from multiprocessing import shared_memory
import time
from harness import (emit_verify_tag, hardware_info, package_stats, partitions,
                     process_counts, save_json, settings, spawn_context, stats_from_samples)


def stage_a(shm_buf, n: int) -> float:
    # Exactly 5% of N elements. Five sin + five sqrt operations per element.
    data = shm_buf.cast("d")
    result = 0.0
    try:
        for i in range(n // 20):
            x = data[i] + 0.01 + (i % 97) * 0.00001
            for _ in range(5):
                x = math.sin(x + 0.1)
                x = math.sqrt(abs(x) + 1.0)
            result += x
    finally:
        del data
    return result


def stage_b_worker(args: tuple) -> float:
    shm_name, n, begin, end = args
    shm = shared_memory.SharedMemory(name=shm_name)
    values = (ctypes.c_double * n).from_buffer(shm.buf)
    result = 0.0
    try:
        for i in range(begin, end):
            x = values[i] + 0.001 + (i % 113) * 0.00001
            # 10 groups of sin, cos, and scalar multiply-add = 30 units.
            for _ in range(10):
                x = math.sin(x) + math.cos(x * 0.5) + x * 0.00001
            result += x
    finally:
        del values
        shm.close()
    return result


def run_case(n: int, p: int, warmup: int, runs: int) -> dict:
    ctx = spawn_context()
    shm = shared_memory.SharedMemory(create=True, size=n * 8)
    args = [(shm.name, n, lo, hi) for lo, hi in partitions(n, p)]
    elapsed = []
    serial_elapsed = []
    checksums = []
    try:
        # Pool creation is explicitly OUTSIDE the timed loop.
        with ctx.Pool(processes=p) as pool:
            for trial in range(warmup + runs):
                t0 = time.perf_counter_ns()
                first = stage_a(shm.buf, n)
                t1 = time.perf_counter_ns()
                # Map exactly p chunks, with no shared writes in Stage B.
                parts = pool.map(stage_b_worker, args, chunksize=1)
                t2 = time.perf_counter_ns()
                if trial >= warmup:
                    elapsed.append((t2 - t0) / 1e6)
                    serial_elapsed.append((t1 - t0) / 1e6)
                    checksums.append(first + sum(parts))
    finally:
        shm.close()
        shm.unlink()
    bench = stats_from_samples(elapsed)
    return {"p": p, "n": n, **package_stats(bench),
            "stage_a_median_ms": stats_from_samples(serial_elapsed).median_ms,
            "checksum": checksums[-1]}


def run(student_id: int, mode: str = "full") -> dict:
    hw = hardware_info()
    cfg = settings(mode)
    physical, logical = process_counts(hw["logical_threads"], hw["physical_cores"], mode)
    strong = []
    for p in range(1, logical + 1):
        case = run_case(cfg["strong_n"], p, cfg["warmup"], cfg["runs"])
        strong.append(case)
        print(f"Strong p={p:>2}: {case['median_ms']:.3f} ms", flush=True)
    t1s = strong[0]["median_ms"]
    for row in strong:
        row["speedup"] = t1s / row["median_ms"]
        row["efficiency"] = row["speedup"] / row["p"]
    serial_fraction = strong[0]["stage_a_median_ms"] / t1s
    amdahl_cap = 1 / serial_fraction if serial_fraction else float("inf")

    weak = []
    for p in range(1, logical + 1):
        case = run_case(p * cfg["weak_n0"], p, cfg["warmup"], cfg["runs"])
        weak.append(case)
        print(f"Weak   p={p:>2}: {case['median_ms']:.3f} ms", flush=True)
    t1w = weak[0]["median_ms"]
    for row in weak:
        p = row["p"]
        row["weak_efficiency"] = t1w / row["median_ms"]
        row["alpha"] = row["stage_a_median_ms"] / row["median_ms"]
        row["gustafson_speedup"] = p - row["alpha"] * (p - 1)
    inflection = None
    if 1 < physical < logical:
        before = strong[physical - 1]["speedup"] - strong[physical - 2]["speedup"]
        after = strong[physical]["speedup"] - strong[physical - 1]["speedup"]
        inflection = {"boundary": [physical, physical + 1],
                      "previous_increment": before, "boundary_increment": after,
                      "delta_increment": after - before}
    tag = emit_verify_tag("PHASE2", student_id, strong[-1]["speedup"], weak[-1]["gustafson_speedup"])
    print(tag)
    return {"phase": 2, "student_id": student_id, "mode": mode,
            "used_physical_cores": physical, "used_logical_threads": logical,
            "strong_n": cfg["strong_n"], "weak_n0": cfg["weak_n0"],
            "strong": strong, "weak": weak, "serial_fraction": serial_fraction,
            "amdahl_cap": amdahl_cap, "inflection": inflection,
            "tag_inputs": [strong[-1]["speedup"], weak[-1]["gustafson_speedup"]],
            "verify_tag": tag}


if __name__ == "__main__":

"""Phase 3: empirical memory/compute ceilings and three double-precision GEMM forms."""
import argparse
import ctypes
import math
from multiprocessing import shared_memory
from harness import (emit_verify_tag, hardware_info, package_stats, partitions,
                     process_counts, run_benchmark, save_json, settings, spawn_context)


def copy_worker(a_name: str, b_name: str, start: int, end: int) -> None:
    a = shared_memory.SharedMemory(name=a_name)
    b = shared_memory.SharedMemory(name=b_name)
    try:
        # Copy equal-length chunks as raw bytes across OS shared memory.
        if end > start:
            ctypes.memmove(ctypes.addressof(ctypes.c_char.from_buffer(a.buf, start)),
                           ctypes.addressof(ctypes.c_char.from_buffer(b.buf, start)),
                           end - start)
    finally:
        a.close()
        b.close()


def compute_worker(loops: int) -> None:
    # Sixteen independent Python scalar multiply-add accumulators.
    # One multiply and one add per assignment = 2 modeled FLOPs.
    x0, x1, x2, x3 = 0.1, 0.2, 0.3, 0.4
    x4, x5, x6, x7 = 0.5, 0.6, 0.7, 0.8
    x8, x9, x10, x11 = 0.9, 1.0, 1.1, 1.2
    x12, x13, x14, x15 = 1.3, 1.4, 1.5, 1.6
    for _ in range(loops):
        x0 = x0 * 0.9999999 + 0.000001
        x1 = x1 * 0.9999999 + 0.000001
        x2 = x2 * 0.9999999 + 0.000001
        x3 = x3 * 0.9999999 + 0.000001
        x4 = x4 * 0.9999999 + 0.000001
        x5 = x5 * 0.9999999 + 0.000001
        x6 = x6 * 0.9999999 + 0.000001
        x7 = x7 * 0.9999999 + 0.000001
        x8 = x8 * 0.9999999 + 0.000001
        x9 = x9 * 0.9999999 + 0.000001
        x10 = x10 * 0.9999999 + 0.000001
        x11 = x11 * 0.9999999 + 0.000001
        x12 = x12 * 0.9999999 + 0.000001
        x13 = x13 * 0.9999999 + 0.000001
        x14 = x14 * 0.9999999 + 0.000001
        x15 = x15 * 0.9999999 + 0.000001
    # Calculated quantity is deliberately used; Python doesn't elide scalar work.
    if not math.isfinite(x0 + x1 + x2 + x3 + x4 + x5 + x6 + x7 + x8 + x9 + x10 + x11 + x12 + x13 + x14 + x15):
        raise ArithmeticError("Non-finite compute result")


def start_and_join(ctx, target, args_list: list[tuple]) -> None:
    jobs = [ctx.Process(target=target, args=args) for args in args_list]
    for job in jobs:
        job.start()
    for job in jobs:
        job.join()
    if any(j.exitcode != 0 for j in jobs):
        raise RuntimeError(f"Worker failed: {target.__name__}")


def measure_bandwidth(p: int, nbytes: int, warmup: int, runs: int) -> dict:
    a = shared_memory.SharedMemory(create=True, size=nbytes)
    b = shared_memory.SharedMemory(create=True, size=nbytes)
    ctx = spawn_context()
    bounds = partitions(nbytes, p)
    args = [(a.name, b.name, lo, hi) for lo, hi in bounds]
    # Ensure source is initialized (SharedMemory is zero-filled on creation).
    try:
        stats = run_benchmark(lambda: start_and_join(ctx, copy_worker, args), warmup, runs)
    finally:
        a.close()
        a.unlink()
        b.close()
        b.unlink()
    bw = (2 * nbytes) / (stats.median_ms / 1000) / 1e9
    return {"buffer_bytes_each": nbytes, "total_traffic_bytes": 2 * nbytes,
            "bandwidth_gbs": bw, **package_stats(stats)}


def measure_compute(p: int, desired_flops: int, warmup: int, runs: int) -> dict:
    ctx = spawn_context()
    loops = math.ceil(desired_flops / 32)
    flops_per_worker = 32 * loops
    args = [(loops,) for _ in range(p)]
    stats = run_benchmark(lambda: start_and_join(ctx, compute_worker, args), warmup, runs)
    return {"flops_per_worker": flops_per_worker, "total_flops": flops_per_worker * p,
            "compute_gflops": (flops_per_worker * p) / (stats.median_ms / 1000) / 1e9,
            **package_stats(stats)}


def gemm_worker(name: str, n: int, row_start: int, row_end: int,
                variant: str, bs: int) -> None:
    shm = shared_memory.SharedMemory(name=name)
    count = n * n
    arr = (ctypes.c_double * (3 * count)).from_buffer(shm.buf)
    ab = 0
    bb = count
    cb = 2 * count
    try:
        if variant == "naive":
            for i in range(row_start, row_end):
                line = i * n
                for j in range(n):
                    acc = 0.0
                    for k in range(n):
                        acc += arr[ab + line + k] * arr[bb + k * n + j]
                    arr[cb + line + j] = acc
        elif variant == "reordered":
            for i in range(row_start, row_end):
                line = i * n
                for k in range(n):
                    aik = arr[ab + line + k]
                    brow = bb + k * n
                    for j in range(n):
                        off = cb + line + j
                        arr[off] += aik * arr[brow + j]
        elif variant == "blocked":
            # 6 loops: tile ii, jj, kk, followed by micro i, k, j.
            for ii in range(row_start, row_end, bs):
                i_end = min(ii + bs, row_end)
                for jj in range(0, n, bs):
                    j_end = min(jj + bs, n)
                    for kk in range(0, n, bs):
                        k_end = min(kk + bs, n)
                        for i in range(ii, i_end):
                            line = i * n
                            for k in range(kk, k_end):
                                aik = arr[ab + line + k]
                                brow = bb + k * n
                                for j in range(jj, j_end):
                                    off = cb + line + j
                                    arr[off] += aik * arr[brow + j]
        else:
            raise ValueError(f"Unknown GEMM variant {variant}")
    finally:
        del arr
        shm.close()


def gemm_case(shm, n: int, p: int, variant: str, bs: int,
              warmup: int, runs: int) -> tuple[dict, list[float]]:
    ctx = spawn_context()
    args = [(shm.name, n, lo, hi, variant, bs) for lo, hi in partitions(n, p)]
    c_offset = 2 * n * n * 8
    output_bytes = n * n * 8
    def trial() -> None:
        # Output clearing is small and included in the trial time.
        ctypes.memset(ctypes.addressof(ctypes.c_char.from_buffer(shm.buf, c_offset)),
                      0, output_bytes)
        start_and_join(ctx, gemm_worker, args)
    stats = run_benchmark(trial, warmup, runs)
    output = list(shm.buf[c_offset:c_offset + n * n * 8].cast("d"))
    gflops = (2 * n ** 3) / (stats.median_ms / 1000.0) / 1e9
    item = {"variant": variant, "bs": bs if variant == "blocked" else None,
            "gflops": gflops, **package_stats(stats)}
    if variant == "blocked":
        item["tile_footprint_bytes"] = 3 * bs * bs * 8
        item["operational_intensity"] = bs / 12.0
    return item, output


def max_error(reference: list[float], actual: list[float]) -> float:
    if len(reference) != len(actual):
        raise ValueError("Different output sizes")
    return max((abs(a - b) for a, b in zip(reference, actual)), default=0.0)


def run(student_id: int, mode: str = "full") -> dict:
    hw = hardware_info()
    cfg = settings(mode)
    p, _ = process_counts(hw["logical_threads"], hw["physical_cores"], mode)
    bw = measure_bandwidth(p, cfg["buffer_bytes"], cfg["warmup"], cfg["runs"])
    print(f"Memory bandwidth: {bw['bandwidth_gbs']:.3f} GB/s", flush=True)
    compute = measure_compute(p, cfg["compute_flops"], cfg["warmup"], cfg["runs"])
    print(f"Scalar compute: {compute['compute_gflops']:.5f} GFLOP/s", flush=True)
    ridge = compute["compute_gflops"] / bw["bandwidth_gbs"] if bw["bandwidth_gbs"] else float("inf")
    n = cfg["gemm_n"]
    shm = shared_memory.SharedMemory(create=True, size=3 * n * n * 8)
    matrix_count = n * n
    try:
        arr = (ctypes.c_double * (3 * matrix_count)).from_buffer(shm.buf)
        for i in range(n):
            for j in range(n):
                arr[i * n + j] = ((i * 13 + j * 7) % 19 - 9) / 19.0
                arr[matrix_count + i * n + j] = ((i * 11 - j * 5) % 23 - 11) / 23.0
        del arr
        variants = []
        naïve, ref = gemm_case(shm, n, p, "naive", 0, cfg["warmup"], cfg["runs"])
        variants.append(naïve)
        print(f"GEMM naive: {naïve['median_ms']:.3f} ms", flush=True)
        reordered, reordered_output = gemm_case(shm, n, p, "reordered", 0, cfg["warmup"], cfg["runs"])
        reordered["max_error_vs_naive"] = max_error(ref, reordered_output)
        variants.append(reordered)
        for bs in cfg["tiles"]:
            entry, actual = gemm_case(shm, n, p, "blocked", bs, cfg["warmup"], cfg["runs"])
            entry["max_error_vs_reordered"] = max_error(reordered_output, actual)

            if entry["max_error_vs_reordered"] >= 1e-9:

    finally:

        shm.unlink()
    if reordered["max_error_vs_naive"] >= 1e-9:

    best = max((v for v in variants if v["variant"] == "blocked"), key=lambda v: v["gflops"])
    tag = emit_verify_tag("PHASE3", student_id, naïve["gflops"], best["gflops"])

    return {"phase": 3, "student_id": student_id, "mode": mode,

            "variants": variants, "best_bs": best["bs"],

            "tag_inputs": [naïve["gflops"], best["gflops"]],

"""Phase 4: skewed workload with block, cyclic, and dynamic chunk scheduling."""
import argparse
import math
import time
from harness import (emit_verify_tag, hardware_info, partitions, process_counts,
                     save_json, settings, spawn_context, stats_from_samples, package_stats)


def work_item(i: int) -> float:
    # Exactly K(i) = 1 + floor((i/40)^2) iterations (using exact integer math).
    count = 1 + (i * i) // 1600
    x = 0.1 + (i % 73) * 0.0001
    for _ in range(count):
        x = math.sin(x) + math.cos(0.5 * x) + math.sqrt(abs(x) + 1.0)
    return x


def scheduling_worker(rank: int, p: int, n: int, method: str,
                      chunk: int, counter, worker_times, checksums) -> None:
    active = 0.0
    total = 0.0
    if method == "block":
        # The partition helper gives contiguous [lo, hi) ranges.
        lo, hi = partitions(n, p)[rank]
        t0 = time.perf_counter()
        for i in range(lo, hi):
            total += work_item(i)
        active += time.perf_counter() - t0
    elif method == "cyclic":
        t0 = time.perf_counter()
        for i in range(rank, n, p):
            total += work_item(i)
        active += time.perf_counter() - t0
    elif method == "dynamic":
        while True:
            # A synchronized mp.Value('i', 0) is the shared chunk queue.
            with counter.get_lock():
                first = counter.value
                counter.value += chunk
            if first >= n:
                break
            t0 = time.perf_counter()
            for i in range(first, min(n, first + chunk)):
                total += work_item(i)
            active += time.perf_counter() - t0
    else:
        raise ValueError(method)
    worker_times[rank] = active
    checksums[rank] = total


def run_case(p: int, n: int, method: str, chunk: int, warmup: int, runs: int) -> dict:
    ctx = spawn_context()
    walls, imbalances, check_values = [], [], []
    worker_time_samples = []
    for trial in range(warmup + runs):
        index = ctx.Value('i', 0)
        worker_times = ctx.Array('d', p, lock=False)
        checksums = ctx.Array('d', p, lock=False)
        workers = [ctx.Process(target=scheduling_worker,
                               args=(rank, p, n, method, chunk, index, worker_times, checksums))
                   for rank in range(p)]
        t0 = time.perf_counter_ns()
        for worker in workers:
            worker.start()
        for worker in workers:
            worker.join()
        ms = (time.perf_counter_ns() - t0) / 1e6
        if any(worker.exitcode != 0 for worker in workers):
            raise RuntimeError(f"Scheduling worker failed for {method} C={chunk}")
        times = list(worker_times)
        mean_time = sum(times) / p
        imbalance = max(times) / mean_time if mean_time else 1.0
        if trial >= warmup:
            walls.append(ms)
            imbalances.append(imbalance)
            worker_time_samples.append(times)
            check_values.append(sum(checksums))
    stats = stats_from_samples(walls)
    return {"method": method, "chunk": chunk if method == "dynamic" else None,
            "n": n, "p": p, **package_stats(stats),
            "lambda_imbalance": stats_from_samples(imbalances).median_ms,
            "worker_times_seconds_last": worker_time_samples[-1],
            "checksum": check_values[-1]}


def run(student_id: int, mode: str = "full") -> dict:
    hw = hardware_info()
    cfg = settings(mode)
    p, _ = process_counts(hw["logical_threads"], hw["physical_cores"], mode)
    n = cfg["sched_n"]
    methods = [("block", 0), ("cyclic", 0)] + [("dynamic", c) for c in cfg["chunks"]]
    entries = []
    for method, chunk in methods:
        entry = run_case(p, n, method, chunk, cfg["warmup"], cfg["runs"])
        entries.append(entry)
        print(f"{method:>7} C={chunk:>3}: {entry['median_ms']:.3f} ms, lambda={entry['lambda_imbalance']:.3f}", flush=True)
    baseline = entries[0]
    for item in entries:
        item["speedup_vs_block"] = baseline["median_ms"] / item["median_ms"]
        if not math.isclose(baseline["checksum"], item["checksum"], rel_tol=1e-10, abs_tol=1e-8):
            raise AssertionError(f"Schedules disagree on total work: {item['method']} {item['chunk']}")
    optimal = min((entry for entry in entries if entry["method"] == "dynamic"),
                  key=lambda entry: entry["median_ms"])
    tag = emit_verify_tag("PHASE4", student_id, baseline["median_ms"], optimal["median_ms"])
    print(tag)
    return {"phase": 4, "student_id": student_id, "mode": mode,
            "used_physical_cores": p, "n": n, "cases": entries,
            "best_chunk": optimal["chunk"], "best_dynamic_median_ms": optimal["median_ms"],
            "tag_inputs": [baseline["median_ms"], optimal["median_ms"]],
            "verify_tag": tag}


if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--student-id", type=int, required=True)
    parser.add_argument("--mode", choices=["full", "quick", "smoke"], default="full")
    parser.add_argument("--output", default="results/phase4.json")
    args = parser.parse_args()
    save_json(args.output, run(args.student_id, args.mode))

def num(x, places: int = 4) -> str:
    if x is None:
        return "N/A"
    if not math.isfinite(float(x)):
        return "N/A"
    return f"{x:.{places}f}"


def md_table(headers: list[str], rows: list[list]) -> list[str]:
    table = ["| " + " | ".join(map(str, headers)) + " |",
             "| " + " | ".join("---" for _ in headers) + " |"]
    for row in rows:
        table.append("| " + " | ".join(str(value) for value in row) + " |")
    return table + [""]


def format_bytes(value) -> str:
    if value is None:
        return "Unavailable"
    return f"{value:,} bytes ({value / 1024:.1f} KiB)"
